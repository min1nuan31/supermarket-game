// --- 視角旋轉變數 --- let isTouching = false; let previousTouchX = 0; let previousTouchY = 0; let cameraYaw = 0; // 左右旋轉角度 let cameraPitch = 0; // 上下旋轉角度 // --- 全螢幕單指/雙指觸控滑動視角 --- window.addEventListener('touchstart', (e) => { // 避開搖桿與 UI 按鈕區域 const touch = e.touches[0]; if (touch.clientX > 180 || touch.clientY < window.innerHeight - 180) { isTouching=true; previousTouchX=touch.clientX; previousTouchY=touch.clientY; } }); window.addEventListener('touchmove', (e)=> { if (!isTouching) return; // 取得主要用於滑動視角的觸控點 for (let touch of e.touches) { if (touch.clientX > 180 || touch.clientY < window.innerHeight - 180) { const deltaX=touch.clientX - previousTouchX; const deltaY=touch.clientY - previousTouchY; // 計算旋轉角度（靈敏度設為 0.005） cameraYaw -=deltaX * 0.005; cameraPitch -=deltaY * 0.005; // 限制上下俯仰角度，避免翻轉過頭 cameraPitch=Math.max(-Math.PI / 3, Math.min(Math.PI / 3, cameraPitch)); previousTouchX=touch.clientX; previousTouchY=touch.clientY; break; } } }); window.addEventListener('touchend', ()=> { isTouching = false; }); window.addEventListener('touchcancel', () => { isTouching = false; }); // --- 更新 animate 函數中的移動與視角邏輯 --- function animate() { requestAnimationFrame(animate); if (!inCheckout) { // 1. 根據目前視角的朝向來計算搖桿移動方向 const speed = 0.08; const forwardX = -Math.sin(cameraYaw); const forwardZ = -Math.cos(cameraYaw); const rightX = Math.cos(cameraYaw); const rightZ = -Math.sin(cameraYaw); // 前後左右位移 camera.position.x += (forwardX * (-moveDir.y) + rightX * moveDir.x) * speed; camera.position.z += (forwardZ * (-moveDir.y) + rightZ * moveDir.x) * speed; // 2. 更新攝影機旋轉視角 const euler = new THREE.Euler(cameraPitch, cameraYaw, 0, 'YXZ'); camera.quaternion.setFromEuler(euler); } renderer.render(scene, camera); }
